LanguagePHP, which runs about 70% of websites (WordPress, Laravel, Drupal, Magento).

Inventory scripts

Windows: browser and PHP versions

$checks = @(
  @{Name='Chrome';  Path="$env:ProgramFiles\Google\Chrome\Application\chrome.exe";         Min='156.0.8078.47'},
  @{Name='Edge';    Path="${env:ProgramFiles(x86)}\Microsoft\Edge\Application\msedge.exe"; Min='156.0.4314.42'},
  @{Name='Firefox'; Path="$env:ProgramFiles\Mozilla Firefox\firefox.exe";                 Min='158.0.1'}
)
foreach ($c in $checks) {
  if (Test-Path $c.Path) {
    $v = (Get-Item $c.Path).VersionInfo.ProductVersion
    $status = if ([version]$v -ge [version]$c.Min) {'PATCHED'} else {'VULNERABLE'}
    "{0,-8} {1,-16} {2}" -f $c.Name, $v, $status
  } else { "{0,-8} not installed" -f $c.Name }
}
# PHP: find php.exe on C: and report the version from file metadata (no execution)
Get-ChildItem C:\ -Recurse -Filter php.exe -ErrorAction SilentlyContinue |
  ForEach-Object { "{0}  {1}" -f $_.FullName, $_.VersionInfo.ProductVersion }

Windows: eMudhra certificates and signed files

# Certificate stores (machine + user)
Get-ChildItem Cert:\LocalMachine\Root, Cert:\LocalMachine\CA, Cert:\CurrentUser\Root, Cert:\CurrentUser\CA |
  Where-Object { $_.Subject -match 'emSign|eMudhra|EM (DV|OV|EV) TLS CA|EM OV CS CA' -or $_.Issuer -match 'emSign|eMudhra' } |
  Select-Object PSParentPath, Subject, Issuer, Thumbprint, NotAfter | Format-List

# Executables signed through the compromised code-signing CAs
Get-ChildItem 'C:\Program Files','C:\Program Files (x86)' -Recurse -Include *.exe,*.dll,*.msi -ErrorAction SilentlyContinue |
  Get-AuthenticodeSignature -ErrorAction SilentlyContinue |
  Where-Object { $_.SignerCertificate.Issuer -match 'EM OV CS CA' } |
  Select-Object Path, Status, @{n='Issuer';e={$_.SignerCertificate.Issuer}}
